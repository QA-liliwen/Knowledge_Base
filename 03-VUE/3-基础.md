| 区块           | 干什么                      |
| ------------ | ------------------------ |
| `<template>` | 写 HTML 结构，决定页面长什么样       |
| <`script`>   | 写 JS 逻辑，决定数据从哪来、点击后做什么\| |
| <`style`>    | 写 CSS 样式，决定颜色、间距、好不好看    |
Template
```html
<template>  
  <nav class="navbar navbar-expand-sm navbar-light border-bottom">  
    <div class="container-fluid">  
      <div class="collapse navbar-collapse" id="topMenuCollapse">  
        <ul class="navbar-nav me-auto">
	        使用v-for循环生成菜单，:key为动态绑定，给每一次循环出来的绑定对应id  
	          <li class="nav-item" v-for="tag in first_tags" :key="tag.id">
			    @click点击事件触发函数，:class动态class绑定            
				    <a class="nav-link" style="margin-right: 20px" href="#" @click.prevent="select_tag(tag)" :class="{active: tag.id === active_tag_id}">{{ tag.name }}</a>  
			  </li>  
        </ul>  
        <ul class="navbar-nav">  
          <li class="nav-item">  
            <a class="nav-link" href="http://127.0.0.1:8000/admin/" target="_blank">后台</a>  
          </li>  
        </ul>  
      </div>  
    </div>  
  </nav>  
</template>
```

script

```html
<script>  
	引入axios库，用于发出http请求  
    import axios from 'axios'  
	Vue组件的固定写法。  
    export default {  
	初始化参数  
        data(){  
            return{  
                first_tags: [],  
                active_tag_id: null,  
            }  
        },  
	mounted生命周期，在页面加载后自动执行  
        mounted:function () {  
			  发起get请求，then存储发起后执行的内容  
            axios.get('http://localhost:8000/get_top_menu/').then(res=>{  
			   保存数据到初始化的参数  
                this.first_tags = res.data.first_tags;  
				默认第一个菜单为选中效果  
                if(this.first_tags.length > 0){  
                    this.select_tag(this.first_tags[0])  
                }  
            })  
        },  
		组件函数  
        methods:{  
			定义了一个函数，传入id  
            select_tag(tag){  
			   将id变为actice的id  
                this.active_tag_id = tag.id;  
				向父组件传递通讯，选中的id和name  
                this.$emit('upTag', {tag_id: tag.id, tag_name: tag.name})  
            }  
        }  
    }  
</script>

</style>
```
