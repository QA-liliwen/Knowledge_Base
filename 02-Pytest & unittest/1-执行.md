```python
# 最常用：跑当前目录下所有测试
python -m pytest

# 详细模式，显示每个用例名（推荐）
python -m pytest -v

# 指定某个文件
python -m pytest test_calc.py

# 指定到具体函数
python -m pytest test_calc.py::test_add

# 关键字筛选：只跑名字里含 "add" 的
python -m pytest -k add

# 组合筛选：含 "add" 但不含 "negative"
python -m pytest -k "add and not negative"

# 第一个失败就停（调试时有用）
python -m pytest -x

# 只跑上次失败的
python -m pytest --lfpytest启动
```
